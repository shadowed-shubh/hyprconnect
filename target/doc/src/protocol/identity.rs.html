1use serde::{Deserialize, Serialize};
2
3// This struct represents one device's identity — the same info
4// your phone broadcasts over UDP every few seconds, and the same
5// info your desktop broadcasts back.
6#[derive(Debug, Clone, Serialize, Deserialize)]
7pub struct Identity {
8    #[serde(rename = "deviceId")]
9    pub device_id: String,
10
11    #[serde(rename = "deviceName")]
12    pub device_name: String,
13
14    #[serde(rename = "deviceType")]
15    pub device_type: String,
16
17    #[serde(rename = "protocolVersion")]
18    pub protocol_version: i32,
19
20    // We never fully confirmed this field ourselves, but the KDE
21    // Connect source guarantees it's sent. default_port() means:
22    // if it's ever missing, assume 1716 instead of crashing.
23    #[serde(rename = "tcpPort", default = "default_port")]
24    pub tcp_port: u16,
25
26    #[serde(rename = "incomingCapabilities", default)]
27    pub incoming_capabilities: Vec<String>,
28
29    #[serde(rename = "outgoingCapabilities", default)]
30    pub outgoing_capabilities: Vec<String>,
31}
32
33fn default_port() -> u16 {
34    1716
35}

37// Everything below this line only runs when you type `cargo test`.

39#[cfg(test)]
40mod tests {
41    use super::*;
42
43    #[test]
44    fn parses_real_phone_identity() {
45        // This is a real body we captured from your phone's broadcast.
46        let json = r#"{
47            "deviceId": "7b2b392fb6f049199028d5eedfbca84f",
48            "deviceName": "OnePlus Nord CE 2 Lite 5G",
49            "protocolVersion": 8,
50            "deviceType": "phone",
51            "incomingCapabilities": ["kdeconnect.battery"],
52            "outgoingCapabilities": ["kdeconnect.battery"]
53        }"#;
54
55        let identity: Identity = serde_json::from_str(json).unwrap();
56
57        assert_eq!(identity.device_id, "7b2b392fb6f049199028d5eedfbca84f");
58        assert_eq!(identity.device_name, "OnePlus Nord CE 2 Lite 5G");
59        assert_eq!(identity.protocol_version, 8);
60        // tcpPort wasn't in this JSON, so it should fall back to 1716.
61        assert_eq!(identity.tcp_port, 1716);
62    }
